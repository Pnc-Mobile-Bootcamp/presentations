// imagine what the code of array.forEach() looks like:

func forEach(someFunc) {
	
	for item in self {
		someFunc(item)
	}

}



// imagine what the code of array.filter() looks like:

func filter(filterFunc) {
	var result = []

	for item in self {
		if filterFunc(item) {
			result.append(item)
		}
	}

	return result
}


// imagine what the code of array.map() looks like:

func map(transformFunc) {
	
	var result = []

	for item in self {
		var newValue = transformFunc(item)
		result.append(newValue)
	}

	return result
	
}



// imagine what the code of array.reduce() looks like:

func reduce(intialSummary, reducerFunc) {
	
	var total = initialSummary

	for item in self {
		total = reducerFunc(total, item)
	}
	
	return total
}
